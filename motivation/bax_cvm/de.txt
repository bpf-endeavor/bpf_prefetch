 *
 * One of the thigns CVM algorithm does is a Treap lookup. This implementation
 * does that in a batch.

/* BAX Framework ----------------------------------------------------------- */

/* The state we maintain per each packet */
typedef struct {
	/* the phase is mandetory */
	int phase[0]; /* to which stage packet belongs */ /* NOTE: BAX moves this out of struct */
	/* ----------------- */
	lookup_partial_state_t lookup_state;

	fp_t rnd_u; /* random number we generated for this packet */

	/* we need following information to adjust the algorithm to the fact
	 * that we are apply things in batch */
	fp_t max_remove_prob; // maximum prob deleted after k_i (?)
	uint8_t count_deleted; // number of nodes delted after k_i
} pkt_state_t;

/* We assume the per packet state type is  is pkt_state_t */
BAX_DECLARE_PKT_STATE_TYPE(pkt_state_t)
BAX_BEGIN_OF_FILE()

#define BAX_NEXT_STAGE(stage) {pstate->phase = stage; continue;}
#define BAX_GET_PSTATE(k) &bs->S[k]

/* Some macros to ease working with a batch */
#define ACTION(act) {batch->actions[BAX_k] = act; BAX_NEXT_STAGE(BAX_DONE);}
#define PASS() ACTION(XDP_PASS)
#define DROP() ACTION(XDP_DROP)
#define TX()   ACTION(XDP_TX)

#define BAX_INIT_BATCH_STATE() BAX_memset(bs, 0, sizeof(BAX_batch_state_t));

/* Use this tag when printing */
#define TAG "BAX CVM:"
#define BPF_GLOBAL_FUNC int __attribute__((noinline))

#define MAX(a,b) (a > b ? a : b)
#define DEFAULT_ACTION() DROP()

/* Some global variables */
/* user-space program initializes the Treap and sets this pointer */
arena_treap_t *treap = NULL;
arena_hist_t *hist = NULL;

/* ``fp_t'' is a fixed-point floating number (since we can not do floating
 * point in eBPF)
 * */
fp_t p;

/* Last timestamp we reported the CVM, we do this on a fixed interval */
static uint64_t last_cvm_report = 0;
#define CVM_REPORT_INTERVAL 5000000000

/* NOTE: enable this flag to report histogram of number of packet active during
 * Treap lookup
 * */
/* #define REPORT_ACTIVE_BATCH_SIZE_DURING_LOOKUP */

/* Helpers */
static inline __attribute__((always_inline))
uint64_t cvm_estimate(void)
{
	return ((uint64_t)treap->used * FP_SCALE) / ((uint64_t)p);
}

static inline __attribute__((always_inline))
void reset(void)
{
	treap_reset(treap);
	p = (1 << 31); // FP_ONE
	last_cvm_report = 0;
}

static inline __attribute__((always_inline))
void report_cvm_value(void)
{
	// report estimate every few seconds
	uint64_t ts = bpf_ktime_get_coarse_ns();
	uint64_t dur = ts - last_cvm_report;
	if (dur > CVM_REPORT_INTERVAL) {
		if (last_cvm_report == 0) {
			last_cvm_report = ts;
		} else {
			bpf_printk(TAG"Size estimate: %lld", cvm_estimate());
#ifdef REPORT_ACTIVE_BATCH_SIZE_DURING_LOOKUP
			/* report histogram of active packets when doing lookup */
			hist_report(hist);
			hist_reset(hist);
#endif
			reset();
			last_cvm_report = ts;
		}
	}
}

/*  Data-structure access operations are Wrapped in a global function to be
 *  evaluated as a separate program to pass the complexity limit of eBPF
 *  */
BPF_GLOBAL_FUNC
subprog_treap_insert(uint32_t k, fp_t u) {
	return treap_insert(treap, &k, u);
}

BPF_GLOBAL_FUNC
subprog_treap_delete(uint32_t key) {
	return treap_delete(treap, &key);
}

BPF_GLOBAL_FUNC
subprog_treap_batch_lookup_p2(uint32_t BAX_k) {
	if (BAX_k >= XDP_MAX_BATCH_SIZE)
		return -2;
	int zero = 0;
	BAX_batch_state_t *bs = bpf_map_lookup_elem(&BAX_batch_state_map, &zero);
	pkt_state_t *pstate = BAX_GET_PSTATE(BAX_k);
	int ret = treap_batch_lookup_p2(treap, NULL, &pstate->lookup_state);
	return ret;
}

BPF_GLOBAL_FUNC
subprog_treap_batch_lookup_p1(uint32_t BAX_k, uint32_t key) {
	if (BAX_k >= XDP_MAX_BATCH_SIZE)
		return -2;
	int zero = 0;
	BAX_batch_state_t *bs = bpf_map_lookup_elem(&BAX_batch_state_map, &zero);
	pkt_state_t *pstate = BAX_GET_PSTATE(BAX_k);
	int ret = treap_batch_lookup_p1(treap, (struct treap_key *)&key,
			&pstate->lookup_state);
	return ret;
}

static __always_inline
void __abort(struct xdp_batch_md *batch, BAX_batch_state_t *bs)
{
	const unsigned short batch_size = batch->size;
	BAX_FOR_ALL(`{
		if (pstate->phase != BAX_DONE)
			DROP();
	}')
	reset();
}

#define __associate_arena() \
{ \
	/* check that user-space program has initilized the Treap */ \
	if (treap == NULL || hist == NULL) { \
		bpf_printk(TAG"Arena initialization issue ..."); \
		/* Make sure Arena is associated with our program */ \
		my_kfunc_reg_arena(&arena); \
		return 0; \
	} \
}

static uint32_t lookup_finished = 0;

/* bbb_ indicates that it is a batch packet processing program */
SEC("xdp")
int bbb_cvm_main(struct xdp_batch_md *batch)
{
	__associate_arena();
	/* bpf_printk("batch size: %d", batch_size); */

	BAX_PROG_BEGIN();
	BAX_INIT_BATCH_STATE();

	lookup_finished = 0;

	cast_kern(treap);

	BAX_STAGE(FIRST, `{
		struct ethhdr *eth = data;
		struct iphdr *ip = (struct iphdr *)(eth + 1);
		struct udphdr *udp = (struct udphdr *)(ip + 1);

		if ((void *)(udp + 1) > data_end) {
			lookup_finished++;
			/* The packet is smaller than we expect */
			PASS();
		}

		if (eth->h_proto != bpf_htons(ETH_P_IP)) {
			lookup_finished++;
			/* We only cound IPv4 packets */
			PASS();
		}

		if (ip->protocol != IPPROTO_UDP) {
			lookup_finished++;
			PASS();
		}

		uint16_t tmp_port = bpf_htons(udp->dest);
		if (!(tmp_port >= 8000 && tmp_port < 8128)) {
			lookup_finished++;
			/* It is not for our application */
			PASS();
		}

		/* this can go under a lookup initialize fine-grained API */
		/* int ret = subprog_treap_batch_lookup_p1(BAX_k, ip->saddr); */
		int ret = treap_batch_lookup_p1(treap, (struct treap_key *)&ip->saddr,
				&pstate->lookup_state);
		if (ret) {
			/* Treap is empty, do not continue lookup */
			pstate->lookup_state.valid = true;
			lookup_finished++;
			BAX_NEXT_STAGE(GEN_RANDOM_VAL);
		}

		BAX_NEXT_STAGE(BATCH_LOOKUP_2);
	}')

	BAX_BREAK_TAIL_CALL()
	/* NOTE: the subprogram should also be associated with arena */
	__associate_arena();

	/* NOTE: this is a logic that does not belong to any stage */
	if (treap->root != NULL) {
		/* Prefetch the root of the Treap */
		P((void *)treap->root);
	}

	// TODO: skip if no packet belongs to the phase
	/* We want to run a stage multiple times, walking the Treap down */
	dnl // BAX_STAGE(BATCH_LOOKUP_2, `') // just decllare the stage
	{
#pragma clang loop unroll(disable)
		for(int k = 0; k < TREAP_MAX_HEIGHT; k++) {
			/* We probably do not need to walk all the levels of graph */
			if (lookup_finished == batch_size)
				break;

			uint32_t cntr = 0;
			BAX_STAGE(BATCH_LOOKUP_2, `{
				cntr++;
				/* int ret = subprog_treap_batch_lookup_p2(BAX_k); */
				int ret = treap_batch_lookup_p2(treap, NULL, &pstate->lookup_state);
				if (ret) {
					lookup_finished++;
					BAX_NEXT_STAGE(BATCH_DELETE_1);
					// check for early termination of stage
					if (lookup_finished == batch_size)
						break;
				}
			}')
#ifdef REPORT_ACTIVE_BATCH_SIZE_DURING_LOOKUP
			hist_record(hist, cntr);
#endif
		}

		if (lookup_finished != batch_size) {
			/* The constant upper bound of loop was not enough */
			__abort(batch, bs);
			return -1;
		}
	}

	BAX_BREAK_TAIL_CALL()
	/* NOTE: the subprogram should also be associated with arena */
	__associate_arena();

	BAX_STAGE(BATCH_DELETE_1, `{
		int ret = treap_delete_node_after_lookup(treap, &pstate->lookup_state);
		switch (ret) {
		case 1:
			BAX_NEXT_STAGE(DELETE_BOOKKEEPING);
			break;
		default:
			BAX_NEXT_STAGE(GEN_RANDOM_VAL);
			break;
		}
	}')

	BAX_STAGE(DELETE_BOOKKEEPING, `') /* Just declare a stage */
	/* TODO: I wanted to traverse the batch in reverse.
	 * I have to break the abstraction
	 * */
	{
		pkt_state_t *pstate, *nstate;
		pstate = BAX_GET_PSTATE(batch_size - 1);
		pstate->max_remove_prob = 0;
		pstate->count_deleted = 0;
		if (bs->phases[batch_size - 1] == DELETE_BOOKKEEPING)
			bs->phases[batch_size - 1] = GEN_RANDOM_VAL;

		// next packet in the batch that belongs to this phase (or the last
		// packet)
		int next = batch_size - 1;

		for (int BAX_k = batch_size - 2; BAX_k > -1; BAX_k--) {
			/* Skip packets that are not for this phase */
			pstate = BAX_GET_PSTATE(BAX_k);
			if (pstate->phase != DELETE_BOOKKEEPING)
				continue;

			nstate = BAX_GET_PSTATE(next);
			arena_treap_node_t *next_n = nstate->lookup_state.node;
			fp_t u = next_n == NULL ? 0 : next_n->priority;
			pstate->max_remove_prob =
				MAX(nstate->max_remove_prob, u);

			pstate->count_deleted = nstate->count_deleted;
			/* if we have found a node, we have deleted it */
			if (next_n != NULL)
				pstate->count_deleted += 1;

			/* next phase is  */
			pstate->phase = GEN_RANDOM_VAL;
			next = BAX_k;
		}
	}

	BAX_STAGE(GEN_RANDOM_VAL, `{
		if (!pstate->lookup_state.valid) {
			bpf_printk("delete faild. this must not happen");
			__abort(batch, bs);
			return -1;
		}

		pstate->rnd_u = fp_random();
		BAX_NEXT_STAGE(BATCH_UPDATE_1);
	}')

	BAX_STAGE(BATCH_UPDATE_1, `{
		int ret;
		fp_t u = pstate->rnd_u;
		if (u > p) {
			DEFAULT_ACTION();
		}

		int free_space = TREAP_MAX_SIZE - treap->used - pstate->count_deleted;
		if (free_space > 0) {
			ret = subprog_treap_insert(*(uint32_t *)pstate->lookup_state.key.data, u);
			if (ret != 0) {
				bpf_printk("failed to insert");
				/* TODO: abort processing the batch ? Early termination not supported yet */
				__abort(batch, bs);
				return -1;
			}
			DEFAULT_ACTION();
		}

		// u < p and |B| = s
		arena_treap_node_t *top = treap_top(treap);
		// Assume we have not deleted the elements after k_i. Would we have a
		// different max probability?
		fp_t max_probability = top->priority;
		fp_t max_removed = pstate->max_remove_prob;
		bool root_already_removed = false;
		if (max_probability < max_removed) {
			root_already_removed = true;
			max_probability = max_removed;
		}

		if (u > max_probability) {
			p = u;
			DEFAULT_ACTION(); // continue;
		} else {
			p = max_probability;

			// we might have already removed the root
			if (!root_already_removed) {
				ret = subprog_treap_delete(*(uint32_t *)top->key.data);
				if (ret != 0) {
					bpf_printk(TAG"failed to replace the node (delete): %d", ret);
					/* TODO: abort the batch */
					__abort(batch, bs);
					return -1;
				}
			}

			ret = subprog_treap_insert(*(uint32_t *)pstate->lookup_state.key.data, u);
			if (ret != 0) {
				bpf_printk(TAG"failed to replace the node (insert): %d", ret);
				/* TODO: abort batch */
				__abort(batch, bs);
				return -1;
			}
		}
		DEFAULT_ACTION();
	}')

	report_tput_batch(batch_size);
	report_cvm_value();
	return 0;
}
