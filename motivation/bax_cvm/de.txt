	/* the phase is mandetory */
	int phase[0]; /* to which stage packet belongs */ /* NOTE: BAX moves this out of struct */

	arena_treap_node_t *node;

	fp_t rnd_u; /* random number we generated for this packet */

	fp_t max_remove_prob; // maximum prob deleted after k_i (?)

	/* key we have extracted */
	struct treap_key key; /* IPv4, key we use to store in the Trep */
	/* uint16_t key_off; /1* offset of key on the packet header *1/ */

/* Use this tag when printing */
#define TAG "BAX CVM:"

#define MAX(a,b) (a > b ? a : b)

/* Some global variables */
/* user-space program initializes the Treap and sets this pointer */
arena_treap_t *treap = NULL;
arena_hist_t *hist = NULL;

/* ``fp_t'' is a fixed-point floating number (since we can not do floating
 * point in eBPF)
 * */
fp_t p;

/* Last timestamp we reported the CVM, we do this on a fixed interval */
static uint64_t last_cvm_report = 0;
#define CVM_REPORT_INTERVAL 5000000000

/* NOTE: enable this flag to report histogram of number of packet active during
 * Treap lookup
 * */
/* #define REPORT_ACTIVE_BATCH_SIZE_DURING_LOOKUP */

/* Helpers */
static inline __attribute__((always_inline))
uint64_t cvm_estimate(void)
{
	return ((uint64_t)treap->used * FP_SCALE) / ((uint64_t)p);
}

static inline __attribute__((always_inline))
void reset(void)
{
	treap_reset(treap);
	p = (1 << 31); // FP_ONE
	last_cvm_report = 0;
}

static inline __attribute__((always_inline))
void report_cvm_value(void)
{
	// report estimate every few seconds
	uint64_t ts = bpf_ktime_get_coarse_ns();
	uint64_t dur = ts - last_cvm_report;
	if (dur > CVM_REPORT_INTERVAL) {
		if (last_cvm_report == 0) {
			last_cvm_report = ts;
		} else {
			bpf_printk(TAG"Size estimate: %lld", cvm_estimate());
#ifdef REPORT_ACTIVE_BATCH_SIZE_DURING_LOOKUP
			/* report histogram of active packets when doing lookup */
			hist_report(hist);
			hist_reset(hist);
#endif
			reset();
			last_cvm_report = ts;
		}
	}
}

/*  insert/delete are wrapped in a global function to be evaluated as a
 *  separate program and help with passing the complexity limit
 *  */
BPF_GLOBAL_FUNC
subprog_treap_insert(uint32_t k, fp_t u) {
	return treap_insert(treap, &k, u);
}

BPF_GLOBAL_FUNC
subprog_treap_delete(uint32_t key) {
	return treap_delete(treap, &key);
}

struct __batch_lookup_2_inner_ctx {
	struct xdp_batch_md *batch;
	BAX_batch_state_t *bs;
	uint16_t batch_size;
	uint32_t lookup_finished;
};

static long __batch_lookup_2_inner(__u32 index, void *_ctx)
{
	struct __batch_lookup_2_inner_ctx *ctx = _ctx;
	struct xdp_batch_md *batch = ctx->batch;
	BAX_batch_state_t *bs = ctx->bs;
	uint16_t batch_size = ctx->batch_size;

	uint32_t cntr = 0;

	BAX_STAGE(BATCH_LOOKUP_2, `{
		/* Farbod: this should not happen because we change the phase */
		/* if (pstate->valid) */
		/* 	continue; */

		/* keep track of number of active packets in the batch */
		cntr++;

		/* All of this can go under a fine-grained API lookup */
		arena_treap_node_t *ptr = pstate->node;
		if (ptr == NULL) {
			/* This key does not exist in the Treap */
			pstate->link = NULL;
			pstate->valid = true;
			ctx->lookup_finished++;
			/* NOTE: BATCH_DELETE_1 removes the node we find, we can skip that
			 * stage since there is nothing.
			 * */
			BAX_NEXT_STAGE(BATCH_DELETE_2);
			// TODO: how to express early termination
		}

		struct treap_key *key = &pstate->key;
		void __arena *ptr_key = (void __arena *)(&ptr->key);
		int64_t cmp = treap_key_cmp(key, (void *)ptr_key);
		if (cmp == 0) {
			/* Found the matching key */
			pstate->valid = true;
			ctx->lookup_finished++;
			BAX_NEXT_STAGE(BATCH_DELETE_1);
			// TODO: how to express early termination
		} else if (cmp < 0) {
			/* Explore the left sub-graph */
			pstate->link = &ptr->left;
			pstate->node = ptr->left;
			P((void *)(ptr->left));
		} else {
			/* Explore the right sub-grph */
			pstate->link = &ptr->right;
			pstate->node = ptr->right;
			P((void *)(ptr->right));
		}
	}')

#ifdef REPORT_ACTIVE_BATCH_SIZE_DURING_LOOKUP
	/* NOTE: track number of active packets (from a full batch size) in different iterations */
	/* bpf_printk("active batch: %d", cntr); */
	// hist_record(hist, cntr);
	unsigned int b = (cntr - HIST_START_NUMBER) / HIST_BUCKET_WIDTH;
	hist->buckets[b]++;
#endif

	/* We probably do not need to walk all the levels of graph */
	if (ctx->lookup_finished == batch_size)
		return 1;
	/* Explore the next level of trie */
	return 0;
}

static inline __attribute__((always_inline))
void __abort(struct xdp_batch_md *batch, BAX_batch_state_t *bs)
{
	const unsigned short batch_size = batch->size;
	BAX_FOR_ALL(`{
		if (pstate->phase != BAX_DONE)
			DROP();
	}')
	reset();
}


/* bbb_ indicates that it is a batch packet processing program */
SEC("xdp")
int bbb_cvm_main(struct xdp_batch_md *batch)
{
	/* check that user-space program has initilized the Treap */
	if (treap == NULL || hist == NULL) {
		bpf_printk("something is wrong in setting up Treap & Arena");
		/* Make sure Arena is associated with our program */
		my_kfunc_reg_arena(&arena);
		return 0;
	}

	/* bpf_printk("batch size: %d", batch_size); */

	/* NOTE: do we need to do this? */
	P((void *)treap);

	BAX_PROG_BEGIN();
	BAX_INIT_BATCH_STATE();

	/* NOTE: this is a logic that does not belong to any stage */
	if (treap->root != NULL) {
		/* Prefetch the root of the Treap */
		P((void *)treap->root);
	}

	uint32_t lookup_finished = 0;

	BAX_STAGE(FIRST, `{
		struct ethhdr *eth = data;
		struct iphdr *ip = (struct iphdr *)(eth + 1);
		struct udphdr *udp = (struct udphdr *)(ip + 1);

		if ((void *)(udp + 1) > data_end) {
			lookup_finished++;
			/* The packet is smaller than we expect */
			PASS();
		}

		if (eth->h_proto != bpf_htons(ETH_P_IP)) {
			lookup_finished++;
			/* We only cound IPv4 packets */
			PASS();
		}

		if (ip->protocol != IPPROTO_UDP) {
			lookup_finished++;
			PASS();
		}

		uint16_t tmp_port = bpf_htons(udp->dest);
		if (!(tmp_port >= 8000 && tmp_port < 8128)) {
			lookup_finished++;
			/* It is not for our application */
			PASS();
		}

		/* uint16_t off = (uint64_t)(&ip->saddr) - (uint64_t)(eth); */
		/* pstate->key_off = off; */
		*(uint32_t *)pstate->key.data = ip->saddr;

		/* Merge LOOKUP_1 with this stage */
		/* BAX_NEXT_STAGE(BATCH_LOOKUP_1); */

		/* this can go under a lookup initialize fine-grained API */
		pstate->link = &treap->root;
		pstate->node = treap->root;
		// pstate->valid = false; /* valid is set to false during init */

		if (treap->root == NULL) {
					pstate->valid = true;
					lookup_finished++;
					BAX_NEXT_STAGE(GEN_RANDOM_VAL);
		}

		BAX_NEXT_STAGE(BATCH_LOOKUP_2);
	}')

	// TODO: skip if no packet belongs to the phase
	/* We want to run a stage multiple times, walking the Treap down */
	{
		// TODO: Abstract bpf_loop
		struct __batch_lookup_2_inner_ctx ctx = {};
		ctx.batch = batch;
		ctx.bs = bs;
		ctx.batch_size = batch_size;
		ctx.lookup_finished = lookup_finished;
		bpf_loop(batch_size * TREAP_MAX_HEIGHT, __batch_lookup_2_inner, &ctx, 0);
		lookup_finished = ctx.lookup_finished;

		if (lookup_finished != batch_size) {
			/* The constant upper bound of loop was not enough */
			__abort(batch, bs);
			return -1;
		}
	}

	/* NOTE: Splitted the program into to subprograms for comlexity reasons */
	bpf_tail_call_static(batch, &prog_arr_map, PROG_2);

	/* we must never reach here */
	return -1;
}

// SPLIT_TO_TAIL_PROG() ------
SEC("xdp")
int bbb_cvm_prog2(struct xdp_batch_md *batch)
{
	/* check that user-space program has initilized the Treap */
	if (treap == NULL || hist == NULL) {
		my_kfunc_reg_arena(&arena);
		return 0;
	}
	BAX_PROG_BEGIN();

	BAX_STAGE(BATCH_DELETE_1, `{
		arena_treap_node_t *n = pstate->node;
		arena_treap_link_t *link = pstate->link;

		/* NOTE: This should not happen */
		if (n == NULL) {
			/* key does not exists */
			BAX_NEXT_STAGE(BATCH_DELETE_2);
		}

		if (n->left == NULL) {
			*link = n->right;
		} else {
			// has left child
			if (n->right == NULL) {
				// has one child (the left child)
				*link = n->left;
			} else {
				// NOTE: batch treap: It seems we can not overlap this
				// because the fixing sub tree becomes a mess
				// (I dont know how to do it)
				// TODO: BAX failed to parse the apostrophy for contraction of
				// "do not". It is a weird limitation

				// We need to move the node down, find the immediate
				// successor (the most left  child of the right sub-tree)
				arena_treap_link_t *leaf_link = NULL;
				arena_treap_node_t *leaf = __get_immediate_succesor(n, &leaf_link);
				if (leaf == NULL) {
					// failed to do it in a bounded size
					// NOTE: batch treap: error missed
					bpf_printk("failed to __get_immediate...");
					pstate->valid = false;
					BAX_NEXT_STAGE(GEN_RANDOM_VAL);
				}

				// swap the node with the leaf and then remove the node
				// .. we know leaf does not have a left, if it does not
				// have a right then just remove the node. if it has a
				// right child then put the right child in place of
				// leaf.
				*leaf_link = leaf->right;
				leaf->right = n->right;
				leaf->left = n->left;
				*link = leaf;
				// node has been removed and everything is almost okay
				// except that moving leaf to the nodes position may
				// have disturbed the heap property
				int ret = __fix_sub_tree_heap_property_down(leaf, link);
				if (ret != 0) {
					// NOTE: batch treap: error missed
					bpf_printk("failed to __fix_sub...");
					pstate->valid = false;
					BAX_NEXT_STAGE(GEN_RANDOM_VAL);
				}
			}
		}
		__treap_free_node(treap, n);
		BAX_NEXT_STAGE(BATCH_DELETE_2);
	}')

	BAX_STAGE(BATCH_DELETE_2, `') /* Just declare a stage */
	/* TODO: I wanted to traverse the batch in reverse.
	 * I have to break the abstraction
	 * */
	{
		pkt_state_t *pstate, *nstate;
		pstate = BAX_GET_PSTATE(batch_size - 1);
		pstate->max_remove_prob = 0;
		pstate->count_deleted = 0;
		if (bs->phases[batch_size - 1] == BATCH_DELETE_2)
			bs->phases[batch_size - 1] = GEN_RANDOM_VAL;

		// next packet in the batch that belongs to this phase (or the last
		// packet)
		int next = batch_size - 1;

		for (int BAX_k = batch_size - 2; BAX_k > -1; BAX_k--) {
			/* Skip packets that are not for this phase */
			pstate = BAX_GET_PSTATE(BAX_k);
			if (pstate->phase != BATCH_DELETE_2)
				continue;

			nstate = BAX_GET_PSTATE(next);
			arena_treap_node_t *next_n = nstate->node;
			fp_t u = next_n == NULL ? 0 : next_n->priority;
			pstate->max_remove_prob =
				MAX(nstate->max_remove_prob, u);

			pstate->count_deleted = nstate->count_deleted;
			/* if we have found a node, we have deleted it */
			if (next_n != NULL)
				pstate->count_deleted += 1;

			/* next phase is  */
			pstate->phase = GEN_RANDOM_VAL;
			next = BAX_k;
		}
	}

	BAX_STAGE(GEN_RANDOM_VAL, `{
		if (!pstate->valid) {
			bpf_printk("delete faild. this must not happen");
			__abort(batch, bs);
			return -1;
		}

		pstate->rnd_u = fp_random();
		BAX_NEXT_STAGE(BATCH_UPDATE_1);
	}')

	BAX_STAGE(BATCH_UPDATE_1, `{
		int ret;
		fp_t u = pstate->rnd_u;
		if (u > p) {
			DEFAULT_ACTION();
		}

		int free_space = TREAP_MAX_SIZE - treap->used - pstate->count_deleted;
		if (free_space > 0) {
			ret = subprog_treap_insert(*(uint32_t *)pstate->key.data, u);
			if (ret != 0) {
				bpf_printk("failed to insert");
				/* TODO: abort processing the batch ? Early termination not supported yet */
				__abort(batch, bs);
				return -1;
			}
			DEFAULT_ACTION();
		}

		// u < p and |B| = s
		arena_treap_node_t *top = treap_top(treap);
		// Assume we have not deleted the elements after k_i. Would we have a
		// different max probability?
		fp_t max_probability = top->priority;
		fp_t max_removed = pstate->max_remove_prob;
		bool root_already_removed = false;
		if (max_probability < max_removed) {
			root_already_removed = true;
			max_probability = max_removed;
		}

		if (u > max_probability) {
			p = u;
			DEFAULT_ACTION(); // continue;
		} else {
			p = max_probability;

			// we might have already removed the root
			if (!root_already_removed) {
				ret = subprog_treap_delete(*(uint32_t *)top->key.data);
				if (ret != 0) {
					bpf_printk(TAG"failed to replace the node (delete): %d", ret);
					/* TODO: abort the batch */
					__abort(batch, bs);
					return -1;
				}
			}

			ret = subprog_treap_insert(*(uint32_t *)pstate->key.data, u);
			if (ret != 0) {
				bpf_printk(TAG"failed to replace the node (insert): %d", ret);
				/* TODO: abort batch */
				__abort(batch, bs);
				return -1;
			}
		}
		DEFAULT_ACTION();
	}')

	// ----
	report_tput_batch(batch_size);
	report_cvm_value();
	return 0;
}
